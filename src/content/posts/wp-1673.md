## 【C++】计算从1~1E10的和





<p><a href="https://blog.csdn.net/chongkuachan8070/article/details/101041752" rel="noreferrer noopener">https://blog.csdn.net/chongkuachan8070/article/details/101041752</a></p>



<pre class="wp-block-syntaxhighlighter-code">// paralle.cpp : 定义控制台应用程序的入口点。
//
 
 
#include "stdafx.h"
#include &lt;omp.h>
#include &lt;stdio.h>
#include &lt;stdlib.h>
#include &lt;windows.h>
#include &lt;time.h>
 
#define NUM_THREADS 4
 
int _tmain(int argc, _TCHAR* argv[])
{
	omp_set_num_threads(NUM_THREADS);
 
	long long sum = 0;
	long long sumtmp[NUM_THREADS];
	clock_t t1 = clock();
 
#pragma omp parallel 
	{
		long i;
		long id = omp_get_thread_num();
		long long temp = 0L;
#pragma omp for
		for(i = 1; i &lt;= 1000000000; i++)
		{
			temp+=i;
		}
		sumtmp[id] = temp;
	}
 
	for(long i = 0; i &lt; NUM_THREADS; i++)
	{
		sum += sumtmp[i];
	}
 
	clock_t t2 = clock();
	printf("sum=%lld\n", sum);
	printf("parallel time = %d\n", (t2-t1));
 
	sum = 0;
	t1 = clock();
	for(long i = 1; i &lt;= 1000000000; i++)
		sum += i;
 
	t2 = clock();
	printf("sum=%lld\n", sum);
	printf("serial time = %d\n", (t2-t1));
 
	system("pause");
	return 0;
}</pre>

    print("%s %s %d" % (str_1, str_2, iterable_iterm))

def main():
    iterable = [1, 2, 3, 4, 5]
    pool = Pool()
    str_1 = "This"

    func = partial(somefunc, str_1, str_2)

if __name__ == "__main__":
